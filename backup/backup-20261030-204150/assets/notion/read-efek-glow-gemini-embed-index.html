💡 Struktur HTML & CSS: Elemen Selalu di Atas Glow

<!-- Stuktur yang menjamin Elemen di Depan dan Glow di Bawah: -->
<div class="glow-wrapper">
  <!-- 1. Glow SELALU DI BAWAH (z-index: 1) -->
  <div class="glow-underlay"></div>

  <!-- 2. Elemen SELALU DI ATAS (z-index: 2, Solid Menutup) -->
  <button class="element-surface">
    <span>Tombol Saya</span>
  </button>
</div>

/* Aturan CSS Sederhana & Pasti: */
.glow-wrapper .glow-underlay {
  position: absolute;
  inset: -6px; /* Meluap ke luar dari bawah */
  z-index: 1;  /* MUTLAK DI BAWAH */
  filter: blur(18px);
  opacity: var(--glow-opacity, 0); /* Hilang total saat leave */
}

.glow-wrapper .element-surface {
  position: relative;
  z-index: 2; /* MUTLAK DI ATAS (Menutup tengahnya 100%) */
  background: #ffffff; /* Solid opaque */
}
